# Ubuntu/Debian reference list. install.sh uses these names automatically
# via lib/pkg.sh when falling back to apt; kept here too for boxes where you
# can't run scripts at all and have to install by hand:
#   grep -v '^#' packages/apt.txt | xargs sudo apt-get install -y
zsh
tmux
vim
git
curl
unzip
fontconfig

# starship may not exist in older Ubuntu/Debian repos; if apt-get install
# starship fails, use the local installer instead:
#   curl -sS https://starship.rs/install.sh | sh -s -- --bin-dir ~/.local/bin -y

# kubectl, k9s, and kubelogin aren't reliably in Ubuntu/Debian's own repos

# installers instead:

#   curl -fsSL https://github.com/derailed/k9s/releases/latest/download/k9s_Linux_amd64.tar.gz | tar -xz -C ~/.local/bin k9s
#   curl -fsSL https://github.com/int128/kubelogin/releases/latest/download/kubelogin_linux_amd64.zip -o /tmp/kubelogin.zip && unzip -oj /tmp/kubelogin.zip kubelogin -d ~/.local/bin
